function setup() {
  createCanvas(800,600);
  noLoop();
  background(0);
  
  for (y=0;y<height;y++)
    for (x=0; x<width;x++){
      clr = (x+y)/(width+height)*255;
      
      midx = width/2
      midy = height/2
      dx = x-midx
      dy = y-midy
      d=sqrt(dx*dx+dy*dy)
      set(x, y, color(255-d, d, clr));
    }
      

  updatePixels();
}
