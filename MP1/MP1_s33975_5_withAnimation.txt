x1 = 300; y1 = 100;
x2 = 500; y2 = 500;
x3 = 100; y3 = 500;
let cx, cy;
counter = 0;

function setup() {
  createCanvas(800,600);
  //noLoop();
  frameRate(30);
  background(0);
  stroke(255, 255, 255);
  point(x1, y1);
  point(x2, y2);
  point(x3 , y3);

  cx = x1; cy = y1;
}

function draw() {
  

  for(i=0;i<200;i++){
    rand = floor(random(0,3));
    switch(rand){
      case 0:
        cx = (cx+x1)/2;
        cy = (cy+y1)/2;
        point(cx, cy);
        break;
      case 1:
        cx = (cx+x2)/2;
        cy = (cy+y2)/2;
        point(cx, cy);
        break;
      default:
        cx = (cx+x3)/2;
        cy = (cy+y3)/2;
        point(cx, cy);
    }
    counter++;
  }
  if(counter >= 30000){
    noLoop();
  }
  
  updatePixels();
}
