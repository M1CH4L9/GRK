function setup() {
  createCanvas(800,600);
  noLoop();
}

function draw() {
  background(200, 200, 255, 255);
  
  for (x = 200; x<=600;x++){
    for (y = 200; y<=400; y++){
      set(x, y, color(100, 50, 30));
    }
  }
  for (x = 0;x<=800;x++){
    for (y = 400;y<=600;y++){
      set(x, y, color(0, 100, 0));
    }
  }
  for (i=0;i<1000;i++){
    randx = floor(random(0, 800));
    randy = floor(random(400, 600));
    randr = floor(random(0, 255));
    randg = floor(random(0, 255));
    randb = floor(random(0, 255));
    set(randx, randy, color(randr, randg, randb));
  }
  for (y=60, w = 0;y<200;y++, w+=2){
    for (x=0;x<w;x++){
      set(400+x, y, color(255, 100, 100));
      set(400-x, y, color(255, 100, 100));
    }
  }

  
  updatePixels();
}
